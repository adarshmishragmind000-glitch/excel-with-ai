rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {
    function signedIn() { return request.auth != null; }
    function isAdmin() {
      return signedIn()
        && get(/databases/$(database)/documents/userRoles/$(request.auth.uid)).data.role == 'admin';
    }

    // Each signed-in user can read only their own role document. Client writes are blocked.
    // Assign the initial admin role manually in Firebase Console.
    match /userRoles/{userId} {
      allow read: if signedIn() && request.auth.uid == userId;
      allow write: if false;
    }

    // Placeholder policy for future course data. Admin-only writes; reads require login.
    match /courses/{courseId} {
      allow read: if signedIn();
      allow create, update, delete: if isAdmin();
    }
    match /enrollmentRequests/{requestId} {
      allow read, update, delete: if isAdmin();
      allow create: if signedIn();
    }
    // Deny all other collections unless an explicit rule is added.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
