
rules_version = '2';
service cloud.firestore {
  match /databases/{database}/documents {

    function signedIn() {
      return request.auth != null;
    }

    function isAdmin() {
      return signedIn()
        && exists(/databases/$(database)/documents/userRoles/$(request.auth.uid))
        && get(/databases/$(database)/documents/userRoles/$(request.auth.uid)).data.role == 'admin';
    }

    // Role documents can only be read by their owner.
    // Client-side role changes are prohibited.
    match /userRoles/{userId} {
      allow read: if signedIn() && request.auth.uid == userId;
      allow write: if false;
    }

    // Student profiles: students can create only their own student profile.
    match /studentProfiles/{userId} {
      allow create: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.uid == request.auth.uid
        && request.resource.data.role == 'student'
        && request.resource.data.studentId is string
        && request.resource.data.name is string
        && request.resource.data.phone is string
        && request.resource.data.contactEmail is string
        && request.resource.data.keys().hasOnly([
          'uid', 'studentId', 'name', 'phone',
          'contactEmail', 'role', 'createdAt'
        ]);

      allow read: if signedIn()
        && (request.auth.uid == userId || isAdmin());

      // Students cannot change their ID, UID or role.
      allow update: if signedIn()
        && request.auth.uid == userId
        && request.resource.data.uid == resource.data.uid
        && request.resource.data.studentId == resource.data.studentId
        && request.resource.data.role == 'student'
        && request.resource.data.diff(resource.data).affectedKeys()
          .hasOnly(['name', 'phone', 'contactEmail']);

      allow delete: if isAdmin();
    }

    match /courses/{courseId} {
      allow read: if signedIn();
      allow create, update, delete: if isAdmin();
    }

    match /enrollmentRequests/{requestId} {
      allow read, update, delete: if isAdmin();
      allow create: if signedIn();
    }

    // All other collections remain denied.
    match /{document=**} {
      allow read, write: if false;
    }
  }
}
